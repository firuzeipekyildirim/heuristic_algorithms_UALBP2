% #1 = Run No, #2 = Fitness, #3 = station sequences separated by commas
\newcommand{\UFlowRun}[3]{%
\begin{figure}[t]
\centering
\begin{tikzpicture}[font=\small]

% 12 istasyonu yerleştir
\foreach[count=\stidx from 1] \seq in {#3}{

  % U-shape koordinatları (sol: S1..S6, sağ: S12..S7)
  \ifnum\stidx<7
    \pgfmathsetmacro{\x}{0}
    \pgfmathsetmacro{\y}{-(\stidx-1)*1.25}
  \else
    \pgfmathsetmacro{\x}{6.2}
    \pgfmathsetmacro{\y}{-(12-\stidx)*1.25}
  \fi

  % Node
  \node[station] (R#1S\stidx) at (\x,\y) {%
    \textbf{Individual #1 -- S\stidx}\\[-1pt]
    \scriptsize \seq
  };
}

% Oklar: S1->S2->...->S12
\foreach \i in {1,...,11}{
  \pgfmathtruncatemacro{\j}{\i+1}
  \draw[flow] (R#1S\i) -- (R#1S\j);
}

\end{tikzpicture}
\caption{U-shaped 12-station assignment for ARC83 (Individual #1, Fitness = #2).}
\label{fig:arc83-ind#1}
\end{figure}
}
